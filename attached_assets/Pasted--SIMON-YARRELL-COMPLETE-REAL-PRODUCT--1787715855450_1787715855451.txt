function verifyProduct(product) {

  if (!product.product_id)
    return "REJECT_PRODUCT";

  if (!product.brand)
    return "REJECT_PRODUCT";

  if (!product.product_name)
    return "REJECT_PRODUCT";

  if (!product.category)
    return "REJECT_PRODUCT";

  if (!product.retailer)
    return "REJECT_PRODUCT";

  if (!product.image_url)
    return "REJECT_PRODUCT";

  if (!product.canonical_product_url)
    return "REJECT_PRODUCT";

  if (product.image_verified !== true)
    return "REJECT_PRODUCT";

  if (product.product_verified !== true)
    return "REJECT_PRODUCT";

  if (imageIsStockPhoto(product.image_url))
    return "REJECT_PRODUCT";

  if (imageIsAIgenerated(product.image_url))
    return "REJECT_PRODUCT";

  if (imageIsPlaceholder(product.image_url))
    return "REJECT_PRODUCT";

  if (imageIsBrandLogo(product.image_url))
    return "REJECT_PRODUCT";

  if (!imageMatchesProduct(product))
    return "REJECT_PRODUCT";

  if (!urlMatchesProduct(product))
    return "REJECT_PRODUCT";

  return "VERIFIED_PRODUCT";
}
